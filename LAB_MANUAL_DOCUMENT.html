import React, { Component } from 'react';

class ExpenseSummaryClass extends Component {
  constructor(props) {
    super(props);
    // Initializing component state
    this.state = {
      count: 0,
      lastUpdated: new Date().toLocaleTimeString(),
    };
    // Binding event handler method
    this.handleIncrement = this.handleIncrement.bind(this);
  }

  handleIncrement() {
    this.setState((prevState) => ({
      count: prevState.count + 1,
      lastUpdated: new Date().toLocaleTimeString(),
    }));
  }

  render() {
    return (
      <div className="class-counter-box">
        <h3>React Class Component Counter</h3>
        <p>Current Count: <strong>{this.state.count}</strong></p>
        <button onClick={this.handleIncrement}>Increment Count</button>
      </div>
    );
  }
}

export default ExpenseSummaryClass;

3.b Fetch Data from an API Using React useEffect Hook Unit 3: Important React Concepts

Objective: Write a React program to fetch asynchronous REST API data on component mount using the useEffect hook with loading and error states.

Source Code

FetchExpenses.jsx 
import React, { useState, useEffect } from 'react';
import axios from 'axios';

export default function FetchExpenses() {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let isMounted = true;
    const apiUrl = 'https://expense-tracker-izat.onrender.com/api/expenses';

    axios.get(apiUrl)
      .then(res => {
        if (isMounted) {
          setData(res.data.data || []);
          setLoading(false);
        }
      })
      .catch(err => {
        if (isMounted) setLoading(false);
      });

    return () => { isMounted = false; };
  }, []); // Empty dependency array ensures execution on initial mount only

  if (loading) return <p>⏳ Loading live expenses from API...</p>;

  return (
    <ul>
      {data.map(item => (
        <li key={item._id}>{item.title} — ₹{item.amount}</li>
      ))}
    </ul>
  );
}

ExpenseForm.jsx 
import React, { useState } from 'react';

export default function ExpenseForm({ onAddExpense }) {
  const [formData, setFormData] = useState({
    title: '',
    amount: '',
    category: 'Food',
  });

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.title || !formData.amount) return alert('Please fill all fields');
    onAddExpense({ ...formData, amount: parseFloat(formData.amount) });
    setFormData({ title: '', amount: '', category: 'Food' });
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        type="text"
        placeholder="Expense Title"
        value={formData.title}
        onChange={e => setFormData({ ...formData, title: e.target.value })}
      />
      <input
        type="number"
        placeholder="Amount (₹)"
        value={formData.amount}
        onChange={e => setFormData({ ...formData, amount: e.target.value })}
      />
      <button type="submit">Submit Transaction</button>
    </form>
  );
}

subqueries.sql 
-- 1. SCALAR SUBQUERY: Find transactions greater than overall average spending
SELECT id, title, amount, category
FROM expenses
WHERE amount > (SELECT AVG(amount) FROM expenses)
ORDER BY amount DESC;

-- 2. DERIVED TABLE SUBQUERY: Categories with total spending > ₹1,000
SELECT cat_summary.category, cat_summary.total_spent
FROM (
    SELECT category, SUM(amount) AS total_spent
    FROM expenses
    GROUP BY category
) AS cat_summary
WHERE cat_summary.total_spent > 1000.00;

-- 3. CORRELATED SUBQUERY: Find highest spending record within each category
SELECT e1.id, e1.title, e1.category, e1.amount
FROM expenses e1
WHERE e1.amount = (
    SELECT MAX(e2.amount)
    FROM expenses e2
    WHERE e2.category = e1.category
);

routes/sqlAnalyticsRoutes.js (Dump Generator) 
import express from 'express';
const router = express.Router();

router.get('/export-dump', async (req, res) => {
  const dumpHeader = `-- ExpenseFlow MySQL Dump Generator\n-- Host: localhost  Database: expenseflow_db\n\nCREATE DATABASE IF NOT EXISTS expenseflow_db;\nUSE expenseflow_db;\n\n`;
  const ddlExpenses = `CREATE TABLE IF NOT EXISTS expenses (id INT AUTO_INCREMENT PRIMARY KEY, title VARCHAR(120), amount DECIMAL(10,2), category VARCHAR(50), date DATE);\n`;
  
  const fullDump = dumpHeader + ddlExpenses;
  
  res.setHeader('Content-Type', 'application/sql');
  res.setHeader('Content-Disposition', 'attachment; filename="expenseflow_dump.sql"');
  res.send(fullDump);
});

export default router;
